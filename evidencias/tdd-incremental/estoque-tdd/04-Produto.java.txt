package br.edu.exemplo;
public class Produto {
private String nome; private int quantidade = 0;
 public Produto(String nome) { this.nome = nome; }
 public String getNome() { return nome; }
 public int getQuantidade() { return quantidade; }
 public void adicionarEstoque(int quantidade) { this.quantidade += quantidade; }
 public void retirarEstoque(int quantidade) { this.quantidade -= quantidade; }
}
